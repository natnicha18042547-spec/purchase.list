// รายการหน่วยนับมาตรฐาน (สำหรับเป็นตัวเลือกแนะนำเวลาพิมพ์)
const SUGGESTED_UNITS = ['กก.', 'กรัม', 'ลิตร', 'มล.', 'ฟอง', 'กำ', 'แพ็ค', 'ขวด', 'เม็ด', 'แผ่น', 'ถุง', 'มัด', 'กล่อง'];

interface Ingredient {
  name: string;
  amount: number;
  unit: string;
}

interface Recipe {
  id: number;
  name: string;
  type: string;
  description: string;
  image: string;
  ingredients: Ingredient[];
}

interface BillItem {
  id: string;
  name: string;
  amount: number;
  unit: string;
}

interface RecipeIngredientInput {
  name: string;
  amount: string;
  unit: string;
}

export default function SaYaiSomApp() {
  const [mounted, setMounted] = useState(false);
  const [activeTab, setActiveTab] = useState('menu');
  const [menuSubFilter, setMenuSubFilter] = useState<'all' | 'selected'>('all');
  const [searchTerm, setSearchTerm] = useState('');
  const [newMasterIngredient, setNewMasterIngredient] = useState('');

  // --- State สำหรับข้อมูลหลัก ---
  const [recipes, setRecipes] = useState<Recipe[]>(DEFAULT_RECIPES);
  const [ingredientsMaster, setIngredientsMaster] = useState<string[]>(DEFAULT_INGREDIENTS_MASTER);
  const [selectedMenus, setSelectedMenus] = useState<Record<string, number>>({});

  // --- โหลดข้อมูลจาก LocalStorage หลัง mount ---
  useEffect(() => {
    const savedRecipes = localStorage.getItem('sys_recipes');
    const savedIngredients = localStorage.getItem('sys_ingredients_master');
    const savedMenus = localStorage.getItem('sys_selected_menus');

    if (savedRecipes) setRecipes(JSON.parse(savedRecipes));
    if (savedIngredients) setIngredientsMaster(JSON.parse(savedIngredients));
    if (savedMenus) setSelectedMenus(JSON.parse(savedMenus));

    setMounted(true);
  }, []);

  // --- บันทึกข้อมูลลง LocalStorage อัตโนมัติเมื่อมีการเปลี่ยนแปลง ---
  useEffect(() => {
    if (mounted) {
      localStorage.setItem('sys_recipes', JSON.stringify(recipes));
    }
  }, [recipes, mounted]);

  useEffect(() => {
    if (mounted) {
      localStorage.setItem('sys_ingredients_master', JSON.stringify(ingredientsMaster));
    }
  }, [ingredientsMaster, mounted]);

  useEffect(() => {
    if (mounted) {
      localStorage.setItem('sys_selected_menus', JSON.stringify(selectedMenus));
    }
  }, [selectedMenus, mounted]);

  // --- States สำหรับระบบบิลและการแก้ไขบิล ---
  const [billItems, setBillItems] = useState<BillItem[]>([]); 
  const [isAddingToBill, setIsAddingToBill] = useState(false);
  const [addBillItemName, setAddBillItemName] = useState('');
  const [addBillItemAmount, setAddBillItemAmount] = useState('');
  const [addBillItemUnit, setAddBillItemUnit] = useState('กก.');

  // --- States สำหรับการ เพิ่ม/แก้ไข สูตรอาหาร ---
  const [isRecipeModalOpen, setIsRecipeModalOpen] = useState(false);
  const [editingRecipeId, setEditingRecipeId] = useState<number | null>(null);
  const [recipeName, setRecipeName] = useState('');
  const [recipeType, setRecipeType] = useState('แกง');
  const [recipeDesc, setRecipeDesc] = useState('');
  const [recipeIngredients, setRecipeIngredients] = useState<RecipeIngredientInput[]>([]);
  const [recipeImage, setRecipeImage] = useState(''); 

  const fileInputRef = useRef<HTMLInputElement>(null);

  // อัปเดตตัวเลือกวัตถุดิบตัวแรกในหน้าระบบบิลให้ล้อตามคลังกลางเสมอ
  useEffect(() => {
    if (ingredientsMaster.length > 0) {
      setAddBillItemName(ingredientsMaster[0]);
    }
  }, [ingredientsMaster]);

  // --- คำนวณบิลเริ่มต้นจากเมนูที่เลือกปรุง ---
  useEffect(() => {
    const totals: Record<string, BillItem> = {};
    Object.entries(selectedMenus).forEach(([id, qty]) => {
      if (qty <= 0) return;
      const recipe = recipes.find(r => r.id === parseInt(id));
      if (!recipe) return;

      recipe.ingredients.forEach((ing: Ingredient) => {
        const key = `${ing.name}-${ing.unit}`;
        if (!totals[key]) {
          totals[key] = { id: `auto-${key}`, name: ing.name, amount: 0, unit: ing.unit };
        }
        totals[key].amount += (Number(ing.amount) * qty);
      });
    });

    const initialItems = Object.values(totals).map((item, idx) => ({
      id: `auto-${idx}-${item.name}`,
      name: item.name,
      amount: item.amount,
      unit: item.unit
    }));
    
    setBillItems(initialItems);
  }, [selectedMenus, recipes]);

  // --- ฟังก์ชันจัดการระบบบิล ---
  const updateBillItemAmount = (id: string, newAmount: number) => {
    if (newAmount < 0) return;
    setBillItems(prev => prev.map(item => item.id === id ? { ...item, amount: newAmount } : item));
  };

  const removeBillItem = (id: string) => {
    setBillItems(prev => prev.filter(item => item.id !== id));
  };

  const handleAddItemToBill = (e: React.FormEvent) => {
    e.preventDefault();
    const nameTrimmed = addBillItemName.trim();
    if (!nameTrimmed) return alert('กรุณาระบุชื่อวัตถุดิบ');
    
    const unitTrimmed = addBillItemUnit.trim() || 'หน่วย';
    const inputAmount = addBillItemAmount.trim() === '' ? 0 : Number(addBillItemAmount);
    const finalAmount = inputAmount < 0 ? 0 : inputAmount;

    const existingIndex = billItems.findIndex(item => item.name === nameTrimmed && item.unit === unitTrimmed);

    if (existingIndex > -1) {
      const updated = [...billItems];
      updated[existingIndex].amount += finalAmount;
      setBillItems(updated);
    } else {
      setBillItems([...billItems, {
        id: `manual-${Date.now()}`,
        name: nameTrimmed,
        amount: finalAmount,
        unit: unitTrimmed
      }]);
    }

    setAddBillItemAmount('');
    setIsAddingToBill(false);
  };

  const handleToggleMenu = (recipeId: number, isChecked: boolean) => {
    setSelectedMenus(prev => ({
      ...prev,
      [recipeId]: isChecked ? 1 : 0
    }));
  };

  const totalSelectedCount = useMemo(() => {
    return Object.values(selectedMenus).filter(v => v > 0).length;
  }, [selectedMenus]);

  // --- ฟังก์ชันแชร์บิลเป็นรูปภาพ ---
  const generateBillImageAndShare = () => {
    if (billItems.length === 0) {
      alert('ไม่มีรายการสินค้าในบิล ไม่สามารถสร้างรูปภาพได้ครับ');
      return;
    }

    const canvas = document.createElement('canvas');
    const ctx = canvas.getContext('2d');
    if (!ctx) return;

    const rowHeight = 35;
    const headerHeight = 160;
    const footerHeight = 80;
    const canvasWidth = 500;
    const canvasHeight = headerHeight + (billItems.length * rowHeight) + footerHeight;

    canvas.width = canvasWidth;
    canvas.height = canvasHeight;

    ctx.fillStyle = '#FAF9F6';
    ctx.fillRect(0, 0, canvasWidth, canvasHeight);

    ctx.fillStyle = '#4A6D3A';
    ctx.fillRect(0, 0, canvasWidth, 15);

    ctx.fillStyle = '#4A6D3A';
    ctx.font = 'bold 24px "Prompt", sans-serif';
    ctx.fillText('ใบสั่งของ: ร้านข้าวแกงสระยายโสม', 40, 60);

    ctx.fillStyle = '#666666';
    ctx.font = '14px "Prompt", sans-serif';
    const dateText = `วันที่ออกบิล: ${new Date().toLocaleDateString('th-TH', { day: 'numeric', month: 'long', year: 'numeric' })}`;
    ctx.fillText(dateText, 40, 90);

    ctx.strokeStyle = '#CCCCCC';
    ctx.setLineDash([5, 5]);
    ctx.beginPath();
    ctx.moveTo(40, 115);
    ctx.lineTo(460, 115);
    ctx.stroke();

    ctx.setLineDash([]);
    ctx.fillStyle = '#888888';
    ctx.font = 'bold 14px "Prompt", sans-serif';
    ctx.fillText('รายการวัตถุดิบเด็ด', 40, 140);
    ctx.fillText('ปริมาณที่ต้องซื้อ', 360, 140);

    let currentY = headerHeight;
    ctx.font = '15px "Prompt", sans-serif';
    billItems.forEach((item, idx) => {
      if (idx % 2 === 0) {
        ctx.fillStyle = '#F2E8DF';
        ctx.fillRect(30, currentY - 20, canvasWidth - 60, rowHeight);
      }

      ctx.fillStyle = '#222222';
      ctx.font = '15px "Prompt", sans-serif';
      ctx.fillText(`${idx + 1}. ${item.name}`, 40, currentY);
      
      ctx.fillStyle = '#000000';
      ctx.font = 'bold 15px "Prompt", sans-serif';
      
      const textAmount = item.amount === 0 ? `- ${item.unit}` : `${item.amount.toLocaleString()} ${item.unit}`;
      ctx.fillText(textAmount, 360, currentY);

      currentY += rowHeight;
    });

    ctx.strokeStyle = '#CCCCCC';
    ctx.setLineDash([5, 5]);
    ctx.beginPath();
    ctx.moveTo(40, currentY - 10);
    ctx.lineTo(460, currentY - 10);
    ctx.stroke();

    ctx.setLineDash([]);
    ctx.fillStyle = '#A7D08C';
    ctx.font = 'italic 12px "Prompt", sans-serif';
    ctx.fillText('ระบบอัตโนมัติคลังหลังบ้าน สระยายโสมแอป', 40, currentY + 25);

    const imageURI = canvas.toDataURL('image/png');
    const link = document.createElement('a');
    link.download = `บิลสั่งของ_สระยายโสม_${new Date().toISOString().slice(0,10)}.png`;
    link.href = imageURI;
    link.click();
    
    alert('ระบบบันทึกไฟล์รูปภาพบิลลงในเครื่องให้แล้วครับ!');
  };

  const handleImageChange = (e: React.ChangeEvent<HTMLInputElement>) => {
    const file = e.target.files?.[0];
    if (file) {
      const reader = new FileReader();
      reader.onloadend = () => {
        setRecipeImage(reader.result as string);
      };
      reader.readAsDataURL(file);
    }
  };

  const openAddRecipeModal = () => {
    setEditingRecipeId(null);
    setRecipeName('');
    setRecipeType('แกง');
    setRecipeDesc('');
    setRecipeIngredients([{ name: ingredientsMaster[0] || '', amount: '', unit: 'กก.' }]);
    setRecipeImage('');
    setIsRecipeModalOpen(true);
  };

  const openEditRecipeModal = (recipe: Recipe) => {
    setEditingRecipeId(recipe.id);
    setRecipeName(recipe.name);
    setRecipeType(recipe.type);
    setRecipeDesc(recipe.description);
    const validIngs = recipe.ingredients.map((ing: Ingredient) => ({ 
      name: ing.name, 
      amount: String(ing.amount), 
      unit: ing.unit 
    }));
    setRecipeIngredients(validIngs.length > 0 ? validIngs : [{ name: ingredientsMaster[0] || '', amount: '', unit: 'กก.' }]);
    setRecipeImage(recipe.image || '');
    setIsRecipeModalOpen(true);
  };

  const handleAddMasterIngredient = (e: React.FormEvent) => {
    e.preventDefault();
    const nameTrimmed = newMasterIngredient.trim();
    if (!nameTrimmed) return;
    if (ingredientsMaster.includes(nameTrimmed)) {
      alert('มีวัตถุดิบชื่อนี้ในระบบคลังกลางอยู่แล้วครับ');
      return;
    }
    setIngredientsMaster([...ingredientsMaster, nameTrimmed]);
    setNewMasterIngredient('');
  };

  const handleRemoveMasterIngredient = (nameToRemove: string) => {
    if (window.confirm(`คุณแน่ใจหรือไม่ที่จะลบ "${nameToRemove}" ออกจากฐานข้อมูลวัตถุดิบ?`)) {
      setIngredientsMaster(ingredientsMaster.filter(ing => ing !== nameToRemove));
    }
  };

  const groupedRecipes = useMemo(() => {
    let filtered = recipes.filter(recipe => 
      recipe.name.toLowerCase().includes(searchTerm.toLowerCase())
    );

    if (activeTab === 'menu' && menuSubFilter === 'selected') {
      filtered = filtered.filter(recipe => (selectedMenus[recipe.id] || 0) > 0);
    }

    const categories = ['แกง', 'ผัด', 'ยำ/ลาบ', 'ทอด', 'ต้ม', 'อื่น ๆ'];
    const groups: Record<string, Recipe[]> = {};
    categories.forEach(cat => { groups[cat] = []; });

    filtered.forEach(recipe => {
      const typeKey = recipe.type === 'ต้ม/พะโล้' ? 'ต้ม' : recipe.type;
      if (groups[typeKey]) {
        groups[typeKey].push(recipe);
      } else {
        if (!groups['อื่น ๆ']) groups['อื่น ๆ'] = [];
        groups['อื่น ๆ'].push(recipe);
      }
    });

    return Object.entries(groups).filter(([, items]) => items.length > 0);
  }, [recipes, searchTerm, activeTab, menuSubFilter, selectedMenus]);

  const handleSaveRecipe = (e: React.FormEvent) => {
    e.preventDefault();
    if (!recipeName.trim()) return alert('กรุณากรอกชื่อเมนูอาหาร');

    const validIngredients = recipeIngredients.filter(ing => ing.name.trim() && ing.amount);
    if (validIngredients.length === 0) return alert('กรุณากรอกรายละเอียดวัตถุดิบ');
    
    if (editingRecipeId !== null) {
      setRecipes(recipes.map(r => r.id === editingRecipeId ? {
        ...r,
        name: recipeName,
        type: recipeType,
        description: recipeDesc || 'สูตรเด็ดประจำร้านข้าวแกงสระยายโสม',
        image: recipeImage,
        ingredients: validIngredients.map(ing => ({ name: ing.name.trim(), amount: Number(ing.amount), unit: ing.unit.trim() || 'หน่วย' }))
      } : r));
      alert('อัปเดตข้อมูลสูตรอาหารเรียบร้อยและบันทึกลงความจำเครื่องแล้วครับ!');
    } else {
      const newRecipe = {
        id: Date.now(),
        name: recipeName,
        type: recipeType,
        description: recipeDesc || 'สูตรเด็ดประจำร้านข้าวแกงสระยายโสม',
        image: recipeImage,
        ingredients: validIngredients.map(ing => ({ name: ing.name.trim(), amount: Number(ing.amount), unit: ing.unit.trim() || 'หน่วย' }))
      };
      setRecipes([...recipes, newRecipe]);
      alert('บันทึกสูตรอาหารใหม่ลงความจำเครื่องเรียบร้อยแล้วครับ!');
    }
    setIsRecipeModalOpen(false);
  };

  const deleteRecipe = (id: number) => {
    if (window.confirm('คุณต้องการยืนยันการลบสูตรอาหารนี้ใช่หรือไม่?')) {
      setRecipes(recipes.filter(r => r.id !== id));
      if (selectedMenus[id]) {
        const updatedSelected = { ...selectedMenus };
        delete updatedSelected[id];
        setSelectedMenus(updatedSelected);
      }
    }
  };

  if (!mounted) {
    return (
      <div className="max-w-md mx-auto min-h-screen bg-[#FAF9F6] flex items-center justify-center">
        <div className="text-center">
          <div className="w-16 h-16 bg-[#4A6D3A] rounded-full flex items-center justify-center text-white shadow-lg mx-auto mb-4 animate-pulse">
            <span className="font-bold text-3xl">S</span>
          </div>
          <p className="text-[#4A6D3A] font-semibold">กำลังโหลด...</p>
        </div>
      </div>
    );
  }

  return (
    <div className="max-w-md mx-auto min-h-screen bg-[#FAF9F6] text-[#4A6D3A] pb-28 relative shadow-md">
      
      {/* HEADER */}
      <header className="p-4 flex justify-between items-center bg-white/80 backdrop-blur-md sticky top-0 z-10 border-b border-gray-100">
        <div className="flex items-center gap-2">
          <div className="w-10 h-10 bg-[#4A6D3A] rounded-full flex items-center justify-center text-white shadow-sm">
            <span className="font-bold text-xl">S</span>
          </div>
          <div>
            <h1 className="text-2xl font-bold tracking-tight text-[#4A6D3A]">ข้าวแกงสระยายโสม</h1>
            <p className="text-[10px] text-gray-400 font-semibold tracking-wider">ระบบจัดการคลัง (แยกแท็บเมนูที่เลือก)</p>
          </div>
        </div>
        <button 
          onClick={() => setActiveTab('inventory')}
          className={`p-2 rounded-full transition-colors flex items-center gap-1 ${activeTab === 'inventory' ? 'bg-[#4A6D3A] text-white' : 'bg-gray-100 text-gray-600 hover:bg-gray-200'}`}
        >
          <Database size={18} />
          <span className="text-xs font-bold px-0.5">คลังวัตถุดิบ</span>
        </button>
      </header>

      <main className="p-4">
        
        {/* แถบค้นหา */}
        {(activeTab === 'menu' || activeTab === 'manage') && (
          <div className="relative mb-4">
            <Search className="absolute left-4 top-3.5 text-gray-400" size={20} />
            <input
              type="text"
              placeholder={activeTab === 'menu' ? "ค้นหาเมนูในหน้านี้..." : "ค้นหาชื่อสูตรอาหาร..."}
              className="w-full pl-12 pr-4 py-3 bg-white border border-gray-200 rounded-2xl shadow-sm text-sm text-gray-700 focus:outline-none"
              value={searchTerm}
              onChange={(e) => setSearchTerm(e.target.value)}
            />
          </div>
        )}

        {/* TAB 1: เมนูวันนี้ */}
        {activeTab === 'menu' && (
          <div className="animate-fadeIn space-y-4">
            <div className="flex flex-col gap-3 bg-white p-3.5 rounded-3xl border border-gray-100 shadow-sm">
              <div className="flex justify-between items-center">
                <h2 className="text-sm font-bold text-gray-700">จัดตารางอาหารประจำวันนี้</h2>
                <span className="bg-[#4A6D3A] text-white text-[11px] px-2.5 py-0.5 rounded-full font-bold">
                  เปิดแล้ว {mounted ? totalSelectedCount : 0} เมนู
                </span>
              </div>
              
              <div className="grid grid-cols-2 gap-2 bg-gray-100 p-1 rounded-2xl">
                <button
                  type="button"
                  onClick={() => setMenuSubFilter('all')}
                  className={`py-2 rounded-xl text-xs font-bold transition-all flex items-center justify-center gap-1.5 ${menuSubFilter === 'all' ? 'bg-[#4A6D3A] text-white shadow-sm' : 'text-gray-500 hover:text-gray-700'}`}
                >
                  <Layers size={14} />
                  เมนูทั้งหมด
                </button>
                <button
                  type="button"
                  onClick={() => setMenuSubFilter('selected')}
                  className={`py-2 rounded-xl text-xs font-bold transition-all flex items-center justify-center gap-1.5 ${menuSubFilter === 'selected' ? 'bg-orange-500 text-white shadow-sm' : 'text-gray-500 hover:text-gray-700'}`}
                >
                  <CheckCircle2 size={14} />
                  ที่เลือกแล้ว ({mounted ? totalSelectedCount : 0})
                </button>
              </div>
            </div>

            {groupedRecipes.length > 0 ? (
              groupedRecipes.map(([category, items]) => (

                        onChange={(e) => {
                          const updated = [...recipeIngredients];
                          updated[index].amount = e.target.value;
                          setRecipeIngredients(updated);
                        }}
                      />
                      
                      {/* ช่องกรอกหน่วยนับ พิมพ์ข้อความอิสระได้ */}
                      <div className="w-16">
                        <input 
                          type="text"
                          placeholder="หน่วย"
                          list={`modal-units-${index}`}
                          className="w-full px-1 py-2 bg-gray-50 border border-gray-200 rounded-xl text-xs text-center focus:outline-none"
                          value={ing.unit}
                          onChange={(e) => {
                            const updated = [...recipeIngredients];
                            updated[index].unit = e.target.value;
                            setRecipeIngredients(updated);
                          }}
                        />
                        <datalist id={`modal-units-${index}`}>
                          {SUGGESTED_UNITS.map((unit, idx) => (
                            <option key={idx} value={unit} />
                          ))}
                        </datalist>
                      </div>

                      {recipeIngredients.length > 1 && (
                        <button 
                          type="button" onClick={() => setRecipeIngredients(recipeIngredients.filter((_, i) => i !== index))}
                          className="text-red-400 p-1"
                        >
                          <X size={16} />
                        </button>
                      )}
                    </div>
                  ))}
                </div>
              </div>

              <button type="submit" className="w-full mt-2 py-3 bg-[#4A6D3A] text-white rounded-xl font-bold text-sm">
                บันทึกข้อมูลสูตรอาหารลงเครื่อง
              </button>
            </form>
          </div>
        </div>
      )}

      {/* Navigation Bar */}
      <nav className="fixed bottom-0 left-0 right-0 max-w-md mx-auto bg-white/90 backdrop-blur-lg border-t border-gray-100 px-6 py-3 flex justify-between items-center rounded-t-[2.5rem] shadow-md z-30">
        <button onClick={() => setActiveTab('menu')} className={`flex flex-col items-center gap-1 ${activeTab === 'menu' ? 'text-[#4A6D3A]' : 'text-gray-300'}`}>
          <div className={`p-2 rounded-2xl ${activeTab === 'menu' ? 'bg-[#A7D08C] text-[#4A6D3A]' : ''}`}><Calendar size={22} /></div>
          <span className="text-[10px] font-bold">เมนูวันนี้</span>
        </button>

        <button onClick={() => setActiveTab('bill')} className={`flex flex-col items-center gap-1 ${activeTab === 'bill' ? 'text-[#4A6D3A]' : 'text-gray-300'}`}>
          <div className={`p-2 rounded-2xl ${activeTab === 'bill' ? 'bg-[#A7D08C] text-[#4A6D3A]' : ''}`}><FileText size={22} /></div>
          <span className="text-[10px] font-bold">บิลสั่งของ</span>
        </button>

        <button onClick={() => setActiveTab('manage')} className={`flex flex-col items-center gap-1 ${activeTab === 'manage' ? 'text-[#4A6D3A]' : 'text-gray-300'}`}>
          <div className={`p-2 rounded-2xl ${activeTab === 'manage' ? 'bg-[#A7D08C] text-[#4A6D3A]' : ''}`}><BookOpen size={22} /></div>
          <span className="text-[10px] font-bold">จัดการสูตร</span>
        </button>
      </nav>

      <style dangerouslySetInnerHTML={{ __html: `
        @keyframes fadeIn {
          from { opacity: 0; transform: translateY(8px); }
          to { opacity: 1; transform: translateY(0); }
        }
        .animate-fadeIn { animation: fadeIn 0.2s ease-out forwards; }
      `}} />
    </div>
  );
}
